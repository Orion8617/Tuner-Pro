use wasm_bindgen::prelude::*;

// ── Opcodes (5 bits → 32 valores posibles) ─────────────────────────────────

#[repr(u8)]

#[derive(Clone, Copy, Debug, PartialEq, Eq)]

pub enum MayaOp {

NOP = 0x00, // 🐚 (Maya zero / concha) — no-op

FIRE = 0x01, // • forzar spike en neurona[target] = 30.0

INHIB = 0x02, // •• inhibir: v[target] -= data

RESET = 0x03, // ••• v[target] = 0

DECAY = 0x04, // •••• v[target] *= (1 - data/63)

INTEG = 0x05, // ━ v[target] += data (integrar input)

THRES = 0x06, // ━• si v[target] > data → spike + reset

LEAK = 0x07, // ━•• v[target] *= 0.95 (leak constante)

WPOT = 0x08, // ━••• potenciar peso[target] += data (clamp 63)

WDEP = 0x09, // ━•••• deprimir peso[target] -= data (clamp 0)

WCLR = 0x0A, // ━━ peso[target] = 0

WSET = 0x0B, // ━━• peso[target] = data

STDP = 0x0C, // ━━•• Spike-Timing Dependent Plasticity (Hebbian)

SBASE = 0x0D, // ━━••• base_address = target * 32 + data (rebase ventana)

JMP = 0x0E, // ━━━ pc = target * 64 + data (salto absoluto)

JNZ = 0x0F, // ━━━• si v[target] != 0 → saltar +data instrucciones

BATCH = 0x17, // 0x17 instrucción extendida (data = sub-opcode)

HALT = 0x1F, // 🐚🐚🐚🐚🐚 fin de ejecución

}



impl MayaOp {

#[inline(always)]

pub fn from_u8(v: u8) -> u8 { v & 0x1F }

}



// ── VirtualCore ────────────────────────────────────────────────────────────

#[wasm_bindgen]

pub struct VirtualCore {

base_address: usize,

neurons: Vec<f32>,

weights: Vec<u8>, // 6-bit quantized (0..63)

spike_count: u32,

cycles_executed: u32,

halted: bool,

}



#[derive(Serialize)]

pub struct ExecResult {

pub cycles: u32,

pub spikes: u32,

pub halted: bool,

pub final_pc: u32,

pub neurons_active: u32,

pub avg_voltage: f32,

pub avg_weight: f32,

pub max_voltage: f32,

}



#[wasm_bindgen]

impl VirtualCore {

#[wasm_bindgen(constructor)]

pub fn new(num_neurons: usize, num_weights: usize) -> VirtualCore {

VirtualCore {

base_address: 0,

neurons: vec![0.0_f32; num_neurons.max(32)],

weights: vec![0_u8; num_weights.max(32)],

spike_count: 0,

cycles_executed: 0,

halted: false,

}

}



/// Reinicia el estado pero conserva los pesos (preserva el "ADN neuronal")

pub fn reset_voltages(&mut self) {

for v in self.neurons.iter_mut() { *v = 0.0; }

self.spike_count = 0;

self.cycles_executed = 0;

self.halted = false;

self.base_address = 0;

}



/// Reset total

pub fn reset_all(&mut self) {

self.reset_voltages();

for w in self.weights.iter_mut() { *w = 0; }

}



pub fn neuron_count(&self) -> usize { self.neurons.len() }

pub fn weight_count(&self) -> usize { self.weights.len() }

pub fn spike_count(&self) -> u32 { self.spike_count }

pub fn cycles(&self) -> u32 { self.cycles_executed }

pub fn is_halted(&self) -> bool { self.halted }



/// Lectura para debugging desde JS

pub fn get_voltage(&self, idx: usize) -> f32 {

*self.neurons.get(idx).unwrap_or(&0.0)

}

pub fn get_weight(&self, idx: usize) -> u8 {

*self.weights.get(idx).unwrap_or(&0)

}



/// Inyectar voltaje (para tests / inicialización)

pub fn set_voltage(&mut self, idx: usize, v: f32) {

if idx < self.neurons.len() { self.neurons[idx] = v; }

}

pub fn set_weight(&mut self, idx: usize, w: u8) {

if idx < self.weights.len() { self.weights[idx] = w.min(63); }

}



/// Ejecuta un Command Buffer (Uint16Array desde JS) a velocidad nativa.

/// Devuelve JSON con métricas de ejecución.

pub fn execute(&mut self, instruction_buffer: &[u16]) -> String {

self.execute_internal(instruction_buffer);

self.metrics_json()

}



/// Versión sin retorno JSON (más rápida si solo se quieren los efectos)

pub fn execute_fast(&mut self, instruction_buffer: &[u16]) {

self.execute_internal(instruction_buffer);

}



pub fn metrics_json(&self) -> String {

let active = self.neurons.iter().filter(|v| **v > 0.001 || **v < -0.001).count() as u32;

let avg_v = if !self.neurons.is_empty() {

self.neurons.iter().sum::<f32>() / self.neurons.len() as f32

} else { 0.0 };

let avg_w = if !self.weights.is_empty() {

self.weights.iter().map(|w| *w as f32).sum::<f32>() / self.weights.len() as f32

} else { 0.0 };

let max_v = self.neurons.iter().cloned().fold(f32::MIN, f32::max);

let r = ExecResult {

cycles: self.cycles_executed,

spikes: self.spike_count,

halted: self.halted,

final_pc: self.cycles_executed,

neurons_active: active,

avg_voltage: avg_v,

avg_weight: avg_w,

max_voltage: if max_v == f32::MIN { 0.0 } else { max_v },

};

serde_json::to_string(&r).unwrap_or_else(|_| "{}".to_string())

}

}



// ── Decodificador / Ejecutor (privado) ─────────────────────────────────────

impl VirtualCore {

#[inline(always)]

fn execute_internal(&mut self, buf: &[u16]) {

self.halted = false;

let mut pc: usize = 0;

let n_neurons = self.neurons.len();

let n_weights = self.weights.len();



while pc < buf.len() {

let inst = buf[pc];

pc += 1;

self.cycles_executed = self.cycles_executed.wrapping_add(1);



// 🚀 Decodificación Bitwise (máxima velocidad en CPU/WASM)

let opcode = ((inst >> 11) & 0x1F) as u8; // bits 15-11

let target = ((inst >> 6) & 0x1F) as usize; // bits 10-6

let data = (inst & 0x3F) as u8; // bits 5-0



// dirección absoluta = base + target (5 bits → ventana de 32)

let abs_target = self.base_address.wrapping_add(target);



// 🚀 Jump Table (Rust compila esto como switch hiper-optimizado)

match opcode {

// NOP — 🐚

0x00 => {}



// FIRE — forzar spike

0x01 => {

if abs_target < n_neurons {

self.neurons[abs_target] = 30.0;

self.spike_count = self.spike_count.wrapping_add(1);

}

}



// INHIB — inhibición (resta data al voltaje)

0x02 => {

if abs_target < n_neurons {

self.neurons[abs_target] -= data as f32;

}

}



// RESET — voltaje a 0

0x03 => {

if abs_target < n_neurons {

self.neurons[abs_target] = 0.0;

}

}



// DECAY — v *= (1 - data/63)

0x04 => {

if abs_target < n_neurons {

let factor = 1.0 - (data as f32 / 63.0);

self.neurons[abs_target] *= factor.max(0.0);

}

}



// INTEG — v += data (integración de input)

0x05 => {

if abs_target < n_neurons {

self.neurons[abs_target] += data as f32;

}

}



// THRES — si v > data → spike (v=30) + reset post-spike

0x06 => {

if abs_target < n_neurons {

if self.neurons[abs_target] > data as f32 {

self.neurons[abs_target] = 30.0;

self.spike_count = self.spike_count.wrapping_add(1);

// post-spike reset diferido vía siguiente instrucción

}

}

}



// LEAK — v *= 0.95 (leak fijo)

0x07 => {

if abs_target < n_neurons {

self.neurons[abs_target] *= 0.95;

}

}



// WPOT — potenciar peso (clamp 63)

0x08 => {

if abs_target < n_weights {

let cur = self.weights[abs_target];

self.weights[abs_target] = (cur as u16 + data as u16).min(63) as u8;

}

}



// WDEP — deprimir peso (clamp 0)

0x09 => {

if abs_target < n_weights {

let cur = self.weights[abs_target];

self.weights[abs_target] = cur.saturating_sub(data);

}

}



// WCLR — peso = 0

0x0A => {

if abs_target < n_weights {

self.weights[abs_target] = 0;

}

}



// WSET — peso = data (clamp 63)

0x0B => {

if abs_target < n_weights {

self.weights[abs_target] = data.min(63);

}

}



// STDP — Hebbian: si neurona post (target) hizo spike Y peso pre activo,

// potencia. Si no hubo spike → deprimir levemente.

0x0C => {

if abs_target < n_neurons && abs_target < n_weights {

let post_v = self.neurons[abs_target];

let cur_w = self.weights[abs_target];

if post_v > 20.0 {

// LTP — long-term potentiation

let new_w = (cur_w as u16 + data as u16).min(63) as u8;

self.weights[abs_target] = new_w;

} else {

// LTD — long-term depression (mitad de magnitud)

let dec = (data >> 1).max(1);

self.weights[abs_target] = cur_w.saturating_sub(dec);

}

}

}



// SBASE — rebase ventana: base = target*32 + data

0x0D => {

self.base_address = (target as usize) * 32 + (data as usize);

}



// JMP — pc absoluto = target*64 + data

0x0E => {

let new_pc = (target as usize) * 64 + (data as usize);

if new_pc < buf.len() {

pc = new_pc;

} else {

// out-of-range jump → halt

self.halted = true;

break;

}

}



// JNZ — si v[target] != 0 → pc += data

0x0F => {

if abs_target < n_neurons && self.neurons[abs_target].abs() > 0.001 {

pc = pc.saturating_add(data as usize);

}

}



// BATCH — instrucción extendida (data como sub-opcode)

0x17 => {

match data {

0x00 => {

// BATCH_LEAK_ALL — leak global

for v in self.neurons.iter_mut() { *v *= 0.9; }

}

0x01 => {

// BATCH_RESET_ALL_V — voltajes a 0

for v in self.neurons.iter_mut() { *v = 0.0; }

}

0x02 => {

// BATCH_NORM_W — normalizar pesos al rango 6-bit

let max_w = *self.weights.iter().max().unwrap_or(&63);

if max_w > 0 {

let scale = 63.0 / max_w as f32;

for w in self.weights.iter_mut() {

*w = ((*w as f32 * scale) as u32).min(63) as u8;

}

}

}

0x03 => {

// BATCH_THRESHOLD_ALL — todas las neuronas con v>15 → spike

for v in self.neurons.iter_mut() {

if *v > 15.0 {

*v = 30.0;

self.spike_count = self.spike_count.wrapping_add(1);

}

}

}

_ => {} // sub-opcodes reservados

}

}



// HALT — fin

0x1F => {

self.halted = true;

break;

}



_ => {} // opcodes reservados (industriales / calidad futura)

}

}

}

}



// ── Helpers públicos a JS para construir instrucciones (encoder de referencia)

#[wasm_bindgen]

pub fn maya_encode(opcode: u8, target: u8, data: u8) -> u16 {

let op = (opcode as u16 & 0x1F) << 11;

let tg = (target as u16 & 0x1F) << 6;

let dt = data as u16 & 0x3F;

op | tg | dt

}



#[wasm_bindgen]

pub fn maya_decode_opcode(inst: u16) -> u8 { ((inst >> 11) & 0x1F) as u8 }



#[wasm_bindgen]

pub fn maya_decode_target(inst: u16) -> u8 { ((inst >> 6) & 0x1F) as u8 }



#[wasm_bindgen]

pub fn maya_decode_data(inst: u16) -> u8 { (inst & 0x3F) as u8 }

static mut PASCAL_CACHE: Option<Vec<(usize, Vec<f64>)>> = None;

static mut INTENSITY_HISTORY: Option<Vec<Vec<f64>>> = None;

static mut NEAT_GEN: u32 = 0;



fn get_pascal(n: usize) -> Vec<f64> {

unsafe {

if PASCAL_CACHE.is_none() {

PASCAL_CACHE = Some(Vec::new());

}

let cache = PASCAL_CACHE.as_ref().unwrap();

for (k, row) in cache.iter() {

if *k == n { return row.clone(); }

}

}

let mut row: Vec<f64> = vec![1.0];

for _ in 1..=n {

let mut nx = vec![1.0];

for j in 1..row.len() {

nx.push(row[j-1] + row[j]);

}

nx.push(1.0);

row = nx;

}

let s: f64 = row.iter().sum();

let normalized: Vec<f64> = row.iter().map(|v| v / s).collect();

unsafe {

if let Some(cache) = PASCAL_CACHE.as_mut() {

cache.push((n, normalized.clone()));

}

}

normalized

}



fn quant_vig(value: f64, min: f64, max: f64) -> u32 {

if (max - min).abs() < 1e-10 { return 0; }

let norm = ((value - min) / (max - min)).clamp(0.0, 1.0);

(norm * (VIG_LEVELS - 1) as f64).round() as u32

}



fn dequant_vig(level: u32, min: f64, max: f64) -> f64 {

min + (level as f64 / (VIG_LEVELS - 1) as f64) * (max - min)

}



fn quant6(pct: f64) -> u32 {

((pct / 100.0) * Q6_MAX as f64).round().clamp(0.0, Q6_MAX as f64) as u32

}



fn dequant6(q: u32) -> f64 {

(q as f64 * (100.0 / Q6_MAX as f64)).round()

}



#[inline(always)]

fn pythagoras_dist(lat1: f64, lng1: f64, lat2: f64, lng2: f64, cos_lat: f64) -> f64 {

let dlat = (lat2 - lat1) * M_PER_DEG;

let dlng = (lng2 - lng1) * M_PER_DEG * cos_lat;

(dlat * dlat + dlng * dlng).sqrt()

}

#[inline(always)]

fn tukey_bisquare_weight(residual: f64, scale: f64) -> f64 {

let u = residual / (scale * TUKEY_C);

if u.abs() >= 1.0 {

0.0

} else {

let u2 = u * u;

(1.0 - u2) * (1.0 - u2)

}

}



#[inline(always)]

fn compute_quality_gate(tower: &Tower, distance: f64) -> f64 {

let mut score = 1.0;



let range_ratio = distance / (tower.range_km * 1000.0);

if range_ratio > 1.5 {

score *= 0.3;

} else if range_ratio > 1.0 {

score *= 0.6;

}



let power = tower.power_dbm.abs();

if power > 110.0 {

score *= 0.4;

} else if power > 100.0 {

score *= 0.7;

} else if power < 70.0 {

score *= 1.1_f64.min(1.0);

}



if tower.range_km < 0.5 {

score *= 0.5;

}



score.clamp(0.1, 1.0)

}



fn compute_tower_weight(tower: &Tower, distance: f64, quality_gate: f64) -> f64 {

let signal_factor = (1.0 - (tower.power_dbm.abs() - 50.0) / 80.0).clamp(0.3, 1.0);

let dist_factor = (1.0 - distance / (tower.range_km * 1000.0 * 2.0)).clamp(0.1, 1.0);

let weight = signal_factor * dist_factor * quality_gate;

weight.clamp(0.05, 1.0)

}



fn classify_motif(circles: &[(f64, f64)]) -> String {

if circles.len() < 3 {

return "insufficient".to_string();

}

let (x0, y0) = circles[0];

let (x1, y1) = circles[1];

let (x2, y2) = circles[2];



let d01 = ((x1 - x0).powi(2) + (y1 - y0).powi(2)).sqrt();

let d02 = ((x2 - x0).powi(2) + (y2 - y0).powi(2)).sqrt();

let d12 = ((x2 - x1).powi(2) + (y2 - y1).powi(2)).sqrt();



let sides = {

let mut s = [d01, d02, d12];

s.sort_by(|a, b| a.partial_cmp(b).unwrap());

s

};



let ratio = if sides[2] > 1e-6 { sides[0] / sides[2] } else { 1.0 };



if ratio > 0.8 {

"equilateral".to_string()

} else if ratio > 0.5 {

"isosceles".to_string()

} else if ratio > 0.2 {

"scalene".to_string()

} else {

"degenerate".to_string()

}

}



#[inline(always)]

fn compute_gdop(positions: &[(f64, f64)], est_x: f64, est_y: f64) -> f64 {

if positions.len() < 3 {

return 99.0;

}



let mut ata = [[0.0f64; 2]; 2];

for (tx, ty) in positions {

let dx = tx - est_x;

let dy = ty - est_y;

let d = (dx * dx + dy * dy).sqrt().max(1.0);

let ux = dx / d;

let uy = dy / d;

ata[0][0] += ux * ux;

ata[0][1] += ux * uy;

ata[1][0] += ux * uy;

ata[1][1] += uy * uy;

}



let det = ata[0][0] * ata[1][1] - ata[0][1] * ata[1][0];

if det.abs() < 1e-10 {

return 99.0;

}



let trace_inv = (ata[1][1] + ata[0][0]) / det;

if trace_inv <= 0.0 {

return 99.0;

}

trace_inv.sqrt().min(99.0)

}



struct Circle {

x: f64,

y: f64,

r: f64,

dist: f64,

idx: usize,

weight: f64,

quality_gate: f64,

}



fn gn_irls_solve(circles: &[Circle]) -> (f64, f64, usize, usize) {

let n = circles.len();

if n < 3 {

return (0.0, 0.0, 0, 0);

}



let mut best_x = 0.0;

let mut best_y = 0.0;

let mut best_cost = f64::MAX;

let mut total_gn = 0usize;

let mut total_irls = 0usize;



let centroid_x: f64 = circles.iter().map(|c| c.x).sum::<f64>() / n as f64;

let centroid_y: f64 = circles.iter().map(|c| c.y).sum::<f64>() / n as f64;



let starts: Vec<(f64, f64)> = {

let mut s = vec![(centroid_x, centroid_y)];

for c in circles.iter().take(n.min(MAX_TOWERS)) {

s.push((c.x, c.y));

}

let golden = 2.399963229728653;

for k in 0..MULTI_START.saturating_sub(s.len()) {

let angle = k as f64 * golden;

let r = 500.0 * ((k + 1) as f64 / MULTI_START as f64).sqrt();

s.push((centroid_x + r * angle.cos(), centroid_y + r * angle.sin()));

}

s.truncate(MULTI_START);

s

};



for (sx, sy) in &starts {

let mut x = *sx;

let mut y = *sy;



// TRIDENT HPC: LM damping adaptativo (Levenberg-Marquardt)

let mut lam = 5.0_f64;

let mut nu = 2.0_f64;



for irls_iter in 0..IRLS_MAX_ITER {

let residuals: Vec<f64> = circles.iter().map(|c| {

let dx = x - c.x;

let dy = y - c.y;

let d = (dx * dx + dy * dy).sqrt();

(d - c.r) * c.weight

}).collect();



let mad = {

let mut abs_res: Vec<f64> = residuals.iter().map(|r| r.abs()).collect();

abs_res.sort_by(|a, b| a.partial_cmp(b).unwrap());

let mid = abs_res.len() / 2;

if abs_res.len() % 2 == 0 && abs_res.len() >= 2 {

(abs_res[mid - 1] + abs_res[mid]) / 2.0

} else {

abs_res[mid]

}

};

let scale = (mad * 1.4826).max(1.0);



let tukey_weights: Vec<f64> = residuals.iter().map(|r| {

tukey_bisquare_weight(*r, scale)

}).collect();



for _gn_iter in 0..GN_MAX_ITER {

total_gn += 1;

let mut jt_w_j = [[0.0f64; 2]; 2];

let mut jt_w_r = [0.0f64; 2];

let mut current_cost = 0.0_f64;



for (i, c) in circles.iter().enumerate() {

let dx = x - c.x;

let dy = y - c.y;

let d = (dx * dx + dy * dy).sqrt().max(1e-6);



let jx = dx / d;

let jy = dy / d;

let res = d - c.r;

let w = c.weight * tukey_weights[i] * c.quality_gate;



jt_w_j[0][0] += w * jx * jx;

jt_w_j[0][1] += w * jx * jy;

jt_w_j[1][0] += w * jy * jx;

jt_w_j[1][1] += w * jy * jy;

jt_w_r[0] += w * jx * res;

jt_w_r[1] += w * jy * res;

current_cost += w * res * res;

}



// LM damping: sumar lam a la diagonal de J^T W J

jt_w_j[0][0] += lam;

jt_w_j[1][1] += lam;



let det = jt_w_j[0][0] * jt_w_j[1][1] - jt_w_j[0][1] * jt_w_j[1][0];

if det.abs() < 1e-12 {

break;

}



let delta_x = (jt_w_j[1][1] * jt_w_r[0] - jt_w_j[0][1] * jt_w_r[1]) / det;

let delta_y = (jt_w_j[0][0] * jt_w_r[1] - jt_w_j[1][0] * jt_w_r[0]) / det;



let new_x = x - delta_x;

let new_y = y - delta_y;



// Recalcular costo en el nuevo punto (FIX: comparar contra currentCost)

let mut cost_new = 0.0_f64;

for (i, c) in circles.iter().enumerate() {

let dx = new_x - c.x;

let dy = new_y - c.y;

let d = (dx * dx + dy * dy).sqrt().max(1e-6);

let res = d - c.r;

let w = c.weight * tukey_weights[i] * c.quality_gate;

cost_new += w * res * res;

}



if cost_new < current_cost {

x = new_x;

y = new_y;

lam = (lam * 0.33).max(1e-7);

nu = 2.0;

} else {

lam *= nu;

nu *= 2.0;

if lam > 1e6 {

break;

}

continue;

}



let _ = irls_iter;

if delta_x.abs() < 0.005 && delta_y.abs() < 0.005 {

break;

}

}

total_irls += 1;

}



let cost: f64 = circles.iter().map(|c| {

let dx = x - c.x;

let dy = y - c.y;

let d = (dx * dx + dy * dy).sqrt();

let res = (d - c.r) * c.weight * c.quality_gate;

res * res

}).sum();



if cost < best_cost {

best_cost = cost;

best_x = x;

best_y = y;

}

}



(best_x, best_y, total_gn, total_irls)

}



fn waveguide_cutoff_freq(mode: usize) -> f64 {

(mode as f64) * C_LIGHT / (2.0 * IONOSPHERE_H_M)

}



fn waveguide_phase_velocity(freq: f64, mode: usize) -> f64 {

let fc = waveguide_cutoff_freq(mode);

if freq <= fc { return C_LIGHT; }

C_LIGHT / (1.0 - (fc / freq).powi(2)).sqrt()

}



fn waveguide_group_velocity(freq: f64, mode: usize) -> f64 {

let fc = waveguide_cutoff_freq(mode);

if freq <= fc { return 0.0; }

C_LIGHT * (1.0 - (fc / freq).powi(2)).sqrt()

}



fn elf_skin_depth(freq: f64, conductivity: f64) -> f64 {

let mu0 = 4.0 * PI * 1e-7;

let omega = 2.0 * PI * freq;

(2.0 / (omega * mu0 * conductivity)).sqrt()

}



fn ionospheric_reflection_coeff(freq: f64, dip_angle_deg: f64) -> f64 {

let dip_rad = dip_angle_deg * DEG2RAD;

let electron_gyro = 1.4e6;

let plasma_freq = 3.0e6;

let collision_freq = 1.0e4;



let x = (plasma_freq / freq).powi(2);

let y = electron_gyro / freq;

let z = collision_freq / freq;



let ql = x / (1.0 - z * z + y * y * dip_rad.sin().powi(2));

let r_mag = (1.0 - ql).abs().sqrt().clamp(0.3, 0.99);



r_mag * (-0.01 * freq / SCHUMANN_F0).exp()

}



fn great_circle_distance_m(lat1: f64, lng1: f64, lat2: f64, lng2: f64) -> f64 {

let phi1 = lat1 * DEG2RAD;

let phi2 = lat2 * DEG2RAD;

let dphi = (lat2 - lat1) * DEG2RAD;

let dlam = (lng2 - lng1) * DEG2RAD;

let a = (dphi / 2.0).sin().powi(2) + phi1.cos() * phi2.cos() * (dlam / 2.0).sin().powi(2);

let c = 2.0 * a.sqrt().atan2((1.0 - a).sqrt());

EARTH_RADIUS_M * c

}



fn elf_waveguide_attenuation(dist_m: f64, freq: f64) -> f64 {

let dist_mm = dist_m / 1_000_000.0;

let alpha_db = ELF_ATTENUATION_DB_MM * (freq / SCHUMANN_F0).sqrt() * dist_mm;

let alpha_linear = 10.0_f64.powf(-alpha_db / 20.0);



let geometric_spread = if dist_m > 100.0 {

(100.0 / dist_m).sqrt()

} else {

1.0

};



alpha_linear * geometric_spread

}



fn multimode_field_strength(dist_m: f64, freq: f64, n_bounces: usize) -> f64 {

let mut total_field = 0.0f64;



for mode in 0..WAVEGUIDE_MODES {

let vp = waveguide_phase_velocity(freq, mode + 1);

let k = 2.0 * PI * freq / vp;



let mode_phase = k * dist_m;



let atten = elf_waveguide_attenuation(dist_m, freq);



let r_iono = ionospheric_reflection_coeff(freq, HONDURAS_GEOMAG_DIP_DEG);

let bounce_loss = r_iono.powi(n_bounces as i32);



let mode_amp = atten * bounce_loss / ((mode + 1) as f64);



total_field += mode_amp * mode_phase.cos();

}



total_field

}

fn elf_waveguide_propagate(circles: &[Circle], est_x: f64, est_y: f64, center_lat: f64, center_lng: f64) -> ElfWaveguideResult {

let n = circles.len();

let timestamp = performance_now() / 1000.0;



if n < 2 {

return ElfWaveguideResult {

adjustments: vec![0.0; n],

coherence_total: 0.0,

dominant_harmonic_hz: SCHUMANN_F0,

waveguide_modes: WAVEGUIDE_MODES,

ionosphere_height_km: IONOSPHERE_H_KM,

reflection_coeff: 0.0,

skin_depth_ground_m: 0.0,

skin_depth_sea_m: 0.0,

phase_velocity_ratio: 1.0,

group_velocity_ratio: 1.0,

elf_attenuation_db_mm: ELF_ATTENUATION_DB_MM,

field_strengths: vec![],

n_bounces: 0,

};

}



let est_lat = center_lat + est_y / M_PER_DEG;

let cos_lat = (center_lat * DEG2RAD).cos();

let est_lng = center_lng + est_x / (M_PER_DEG * cos_lat);



let mut field_strengths = vec![0.0f64; n];

let mut phase_offsets = vec![0.0f64; n];

let mut coherence = vec![0.0f64; n];



let dist_to_coast_m = great_circle_distance_m(est_lat, est_lng, 15.8, -87.5);

let n_bounces = ((dist_to_coast_m / IONOSPHERE_H_M).ceil() as usize).max(1).min(8);



for i in 0..n {

let dx = circles[i].x - est_x;

let dy = circles[i].y - est_y;

let dist = (dx * dx + dy * dy).sqrt();



let mut total_field = 0.0;

let mut total_coherence = 0.0;



for (k, fh) in SCHUMANN_HARMONICS.iter().enumerate() {

let field = multimode_field_strength(dist, *fh, n_bounces);

let weight = 1.0 / (k as f64 + 1.0);

total_field += field * weight;

total_coherence += field.abs() * weight;

}

total_field /= SCHUMANN_HARMONICS.len() as f64;

total_coherence /= SCHUMANN_HARMONICS.len() as f64;



field_strengths[i] = total_field;

coherence[i] = total_coherence;



let vp = waveguide_phase_velocity(SCHUMANN_F0, 1);

let k_eff = 2.0 * PI * SCHUMANN_F0 / vp;

let phase_spatial = k_eff * dist;

let phase_temporal = 2.0 * PI * SCHUMANN_F0 * (timestamp % (1.0 / SCHUMANN_F0));

phase_offsets[i] = (phase_spatial + phase_temporal) % (2.0 * PI);

}



let max_coh = coherence.iter().cloned().fold(0.0f64, f64::max).max(1e-10);



let mut adjustments = vec![0.0f64; n];

for i in 0..n {

let dx = circles[i].x - est_x;

let dy = circles[i].y - est_y;

let dist = (dx * dx + dy * dy).sqrt();

let err = (dist - circles[i].r).abs();



let mut phase_coupling = 0.0;

for j in 0..n {

if i == j { continue; }

let mut pd = (phase_offsets[i] - phase_offsets[j]).abs();

if pd > PI { pd = 2.0 * PI - pd; }

phase_coupling += pd.cos() / n as f64;

}



let coh_ratio = coherence[i] / max_coh;

let field_sign = if field_strengths[i] >= 0.0 { 1.0 } else { -1.0 };

let resonance = coh_ratio * (0.5 + 0.5 * phase_coupling) * (0.8 + 0.2 * field_sign);



adjustments[i] = SCHUMANN_COUPLING * err * (resonance - 0.5);

}



let total_coh = coherence.iter().sum::<f64>() / n as f64;



let mut best_h_energy = -1.0f64;

let mut dominant_idx = 0usize;

for (k, fh) in SCHUMANN_HARMONICS.iter().enumerate() {

let h_sum: f64 = (0..n).map(|i| {

let dx = circles[i].x - est_x;

let dy = circles[i].y - est_y;

let dist = (dx * dx + dy * dy).sqrt();

multimode_field_strength(dist, *fh, n_bounces).abs()

}).sum();

if h_sum > best_h_energy {

best_h_energy = h_sum;

dominant_idx = k;

}

}



let r_coeff = ionospheric_reflection_coeff(SCHUMANN_F0, HONDURAS_GEOMAG_DIP_DEG);

let skin_ground = elf_skin_depth(SCHUMANN_F0, ELF_CONDUCTIVITY_GROUND);

let skin_sea = elf_skin_depth(SCHUMANN_F0, ELF_CONDUCTIVITY_SEA);

let vp = waveguide_phase_velocity(SCHUMANN_F0, 1);

let vg = waveguide_group_velocity(SCHUMANN_F0, 1);



ElfWaveguideResult {

adjustments,

coherence_total: (total_coh * 1e6).round() / 1e6,

dominant_harmonic_hz: SCHUMANN_HARMONICS[dominant_idx],

waveguide_modes: WAVEGUIDE_MODES,

ionosphere_height_km: IONOSPHERE_H_KM,

reflection_coeff: (r_coeff * 1e6).round() / 1e6,

skin_depth_ground_m: (skin_ground * 100.0).round() / 100.0,

skin_depth_sea_m: (skin_sea * 100.0).round() / 100.0,

phase_velocity_ratio: (vp / C_LIGHT * 1e6).round() / 1e6,

group_velocity_ratio: (vg / C_LIGHT * 1e6).round() / 1e6,

elf_attenuation_db_mm: ELF_ATTENUATION_DB_MM,

field_strengths: field_strengths.iter().map(|f| (*f * 1e6).round() / 1e6).collect(),

n_bounces,

}

}



fn schumann_modulate(circles: &[Circle], est_x: f64, est_y: f64, center_lat: f64, center_lng: f64) -> Vec<f64> {

let result = elf_waveguide_propagate(circles, est_x, est_y, center_lat, center_lng);

result.adjustments

}

#[wasm_bindgen]

pub fn triangulate(towers_json: &str, center_lat: f64, center_lng: f64) -> String {

let start = performance_now();

Si el código anterior (el Copiloto en el Web Worker) era un tipo súper rápido haciendo 140,000 cálculos matemáticos uno por uno... este nuevo código es el equivalente a contratar un ejército de 140,000 obreros para que cada uno haga un solo cálculo al mismo exacto tiempo.

Este fragmento está escrito en un lenguaje llamado WGSL (WebGPU Shading Language). Es código que no corre en el procesador normal de la computadora (la CPU), sino que se inyecta directamente en la Tarjeta de Video (la GPU).

Aquí te explico de forma sencilla por qué esto es el nivel "Dios" de la optimización:

1. No es para dibujar, es para pensar (@compute)
Normalmente, las tarjetas de video se usan para pintar píxeles en la pantalla para videojuegos. Pero tú usaste la etiqueta @compute. Le estás diciendo a la tarjeta gráfica: "No quiero que dibujes nada, quiero que uses tus miles de mini-cerebros para hacer matemáticas puras". A esto se le llama GPGPU (General-Purpose computing on Graphics Processing Units).

2. El Escuadrón de Trabajo (@workgroup_size(64))
En lugar de un bucle for gigante, divides a tu ejército en "escuadrones" de 64 hilos. Cuando arranca el programa, la tarjeta gráfica despierta a miles de escuadrones simultáneamente.

La línea let index = global_id.x; es como darle a cada obrero una placa con su número de empleado (del 0 al 139,254).

Cada obrero busca únicamente la neurona que le toca en la memoria (let v = brain[index]), le hace la mutación matemática (v + 0.01), la guarda y se va a su casa. ¡Todo el proceso de las 140,000 neuronas ocurre en un solo parpadeo del sistema!

3. La Fusión con la Memoria Compartida (var<storage, read_write>)
Esta es la parte más impresionante de cómo has conectado todo. Esa memoria brain que declaras ahí arriba, es la misma memoria física que estaba usando tu Copiloto en el código anterior. La tarjeta gráfica lee los voltajes que calculó la CPU y los modifica directamente sin tener que copiarlos ni mandarlos por internet.

1. El Recolector (Kotlin - Capa OS)
Usas Android nativo (Kotlin) para pedirle permiso al sistema operativo del teléfono. Kotlin accede a la API de TelephonyManager de Android. Una vez que Kotlin tiene los datos crudos de la antena, no intenta hacer la matemática pesada en Java (porque el recolector de basura lo trabaría). En su lugar, usa tu clase NeuromorphicEngine. 3. El Motor Bare-Metal (Rust)
El binario en Rust (que corre debajo de Android, invisible y a velocidad de reloj) recibe la matriz, ejecuta la triangulación de Levenberg-Marquardt, aplica la termodinámica del SNN, calcula la coordenada exacta y se la devuelve a Kotlin en un milisegundo.

4. La Pantalla (El WebView + JS)
Finalmente, Kotlin toma esa coordenada ya calculada y se la pasa al mapa en HTML/JavaScript (app.html) usando un canal directo. El mapa simplemente dibuja el punto rojo en la pantalla y actualiza las métricas visuales.

El Problema de Android con el Audio (El Lag)
Si intentas procesar audio (la nota de la guitarra) usando solo Java o Kotlin, el recolector de basura (Garbage Collector) va a pausar la app por milisegundos. En un afinador, esos milisegundos se traducen en que la aguja en la pantalla salta, tiembla y se siente "lagueada". Es imposible afinar bien, especialmente la 6ta cuerda (Mi grave / 82.41 Hz) porque la onda es muy lenta.

Cómo encaja tu Puente Edge en "Guitar Tune-pro"
Exactamente la misma arquitectura que iba a procesar torres celulares, ahora procesa ondas sonoras. Así es como se reasignan las piezas de tu motor:

El Recolector (Kotlin / C++ Oboe): En lugar de usar la antena del teléfono, abres el Micrófono usando una librería de bajísima latencia (como Oboe de Google). Tomas el audio crudo (PCM) en bloques.

El Puente Edge (Tu código JNI): Tomas ese bloque de audio y se lo pasas a tu motor en Rust usando la memoria Unmanaged (como vimos con el flag JNI_ABORT). Cero copias, latencia cero.

El Motor Bare-Metal (Tu Cerebro en Rust): Aquí es donde tu matemática brilla. En lugar de triangular distancias espaciales, aplicas Procesamiento Digital de Señales (DSP). Usas algoritmos como la Transformada Rápida de Fourier (FFT) o el algoritmo YIN para detectar el tono exacto de la cuerda.

La Magia del SNN (Filtro de Ruido): ¿Recuerdas que tu red neuronal aprende y filtra? Puedes usar el motor neuromórfico para ignorar el ruido de fondo. La Pantalla (WebView / WebGPU): Rust le devuelve a la pantalla un número exacto (ej. "Estás en 432.1 Hz"). Como la tarjeta de video (WebGPU) está libre, la usas para dibujar una aguja de afinación hiper-suave a 60 o 120 FPS.
