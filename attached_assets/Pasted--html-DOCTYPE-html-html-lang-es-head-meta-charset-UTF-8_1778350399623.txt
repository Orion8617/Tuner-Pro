        // ═══════════════════════════════════════════════════════════
        // 🚀 LEVENBERG-MARQUARDT TRILATERATION ENGINE (ROBUST 2-PASS)
        // ═══════════════════════════════════════════════════════════
        class LMEngine {
            // 🚀 TOQUE FINAL: Geometric Dilution of Precision O(N)
            static calculateGDOP(sensors, p) {
                let a = 0, b = 0, d = 0;
                for (let i = 0; i < sensors.length; i++) {
                    let dx = p.x - sensors[i].x;
                    let dy = p.y - sensors[i].y;
                    let dist = Math.sqrt(dx*dx + dy*dy) || 0.0001;
                    let u = dx / dist; // Unit vector X
                    let v = dy / dist; // Unit vector Y
                    
                    // H^T * H Accumulation
                    a += u * u;
                    b += u * v;
                    d += v * v;
                }
                
                // det(H^T H) = ad - bc (where c = b)
                let det = (a * d) - (b * b);
                
                if (Math.abs(det) < 1e-10) return 99.0; // Singular / Collinear
                
                // Trace of Inverse: (a + d) / det
                let traceInv = (a + d) / det;
                if (traceInv < 0) return 99.0;
                
                return Math.sqrt(traceInv);
            }

            // 🚀 ARTE A COSTO CERO: Clasificador Geométrico O(1)
            static classifyGeometry(sensors) {
                if (sensors.length < 3) return { shape: 'Unknown', quality: 0.0, ratio: 0 };
                
                // Extraer top-3 APs (ya podados y limpios de outliers)
                let s0 = sensors[0], s1 = sensors[1], s2 = sensors[2];
                
                let d01 = Math.sqrt(Math.pow(s0.x - s1.x, 2) + Math.pow(s0.y - s1.y, 2));
                let d02 = Math.sqrt(Math.pow(s0.x - s2.x, 2) + Math.pow(s0.y - s2.y, 2));
                let d12 = Math.sqrt(Math.pow(s1.x - s2.x, 2) + Math.pow(s1.y - s2.y, 2));

                let sides = [d01, d02, d12].sort((a, b) => a - b);
                let ratio = sides[2] / (sides[0] || 0.0001); // d_max / d_min

                if (ratio < 2.0) return { shape: 'Isosceles', quality: 1.0, ratio };
                if (ratio <= 4.0) return { shape: 'Scalene', quality: 0.6, ratio };
                return { shape: 'Cross', quality: 0.3, ratio };
            }

            // 🚀 ARTE A COSTO CERO: Acoplamiento Bilateral de Errores (Opposite Sides)
            static applyBilateralCoupling(sensors, distances, p_est) {
                let n = sensors.length;
                let adj = new Array(n).fill(0);
                let err = new Array(n).fill(0);
                let theta = new Array(n).fill(0);
                const kappa = 0.3; // Coupling constant

                for (let i = 0; i < n; i++) {
                    let dx = p_est.x - sensors[i].x;
                    let dy = p_est.y - sensors[i].y;
                    err[i] = Math.sqrt(dx*dx + dy*dy) - distances[i];
                    theta[i] = Math.atan2(dy, dx);
                }

                let coupledPairs = 0;

                for (let i = 0; i < n; i++) {
                    for (let j = i + 1; j < n; j++) {
                        let dTheta = Math.abs(theta[i] - theta[j]);
                        if (dTheta > Math.PI) dTheta = 2 * Math.PI - dTheta;

                        if (dTheta > 2.5) { // APs en lados opuestos (> 143 grados)
                            let avg_err = (err[i] + err[j]) / 2.0;
                            adj[i] += kappa * (avg_err - err[i]);
                            adj[j] += kappa * (avg_err - err[j]);
                            coupledPairs++;
                        }
                    }
                }

                // Ajustamos las mediciones de distancia originales con el offset acoplado
                let coupledDistances = distances.map((d, i) => d + adj[i]);
                return { coupledDistances, coupledPairs };
            }

            // 🚀 ARTE A COSTO CERO: Cuantización Base-20 Maya (Compresión a 5-bits)
            static quantizeBase20(val, min = -10, max = 10) {
                let norm = (val - min) / (max - min);
                let level = Math.max(0, Math.min(19, Math.floor(norm * 20))); // 0 a 19 cabe en 5 bits
                let dequant = min + ((level + 0.5) / 20.0) * (max - min);
                let error = Math.abs(val - dequant);
                return { level, dequant, error };
            }

            // Pasada de resolución estándar LM
            static solvePass(sensors, distances, initialGuess = {x: 0, y: 0}) {
                let p = { x: initialGuess.x, y: initialGuess.y };
                let lambda = 0.1;
                const maxIter = 50;
                const tolerance = 0.01; 
                let normDp = 999;
                let iter = 0;

                for (iter = 0; iter < maxIter; iter++) {
                    let r = []; 
                    let J = []; 
                    
                    for (let i = 0; i < sensors.length; i++) {
                        let s = sensors[i];
                        let dx = p.x - s.x;
                        let dy = p.y - s.y;
                        let dist = Math.sqrt(dx*dx + dy*dy) || 0.0001;
                        
                        r.push(dist - distances[i]);
                        J.push([dx/dist, dy/dist]);
                    }

                    let JTJ = [[0, 0], [0, 0]];
                    let JTr = [0, 0];

                    for (let i = 0; i < sensors.length; i++) {
                        JTJ[0][0] += J[i][0] * J[i][0];
                        JTJ[0][1] += J[i][0] * J[i][1];
                        JTJ[1][0] += J[i][1] * J[i][0];
                        JTJ[1][1] += J[i][1] * J[i][1];

                        JTr[0] += J[i][0] * r[i];
                        JTr[1] += J[i][1] * r[i];
                    }

                    JTJ[0][0] += lambda;
                    JTJ[1][1] += lambda;

                    let det = JTJ[0][0]*JTJ[1][1] - JTJ[0][1]*JTJ[1][0];
                    if (Math.abs(det) < 1e-10) break; 

                    let invJTJ = [
                        [JTJ[1][1]/det, -JTJ[0][1]/det],
                        [-JTJ[1][0]/det, JTJ[0][0]/det]
                    ];

                    let dpX = invJTJ[0][0]*JTr[0] + invJTJ[0][1]*JTr[1];
                    let dpY = invJTJ[1][0]*JTr[0] + invJTJ[1][1]*JTr[1];

                    p.x -= dpX;
                    p.y -= dpY;

                    normDp = Math.sqrt(dpX*dpX + dpY*dpY);
                    if (normDp < tolerance) break;
                    lambda = Math.max(lambda / 2, 1e-6);
                }
                return { p, iter: iter+1, lambda, error: normDp };
            }

            // 🚀 TOQUE FINAL: Robust NLS with RANSAC-style pruning
            static solveRobust(sensors, distances) {
                // PASS 1: Resolución con todos los sensores (incluyendo valores atípicos)
                let pass1 = this.solvePass(sensors, distances);

                // Calcular residual absoluto: r_i = |‖p̂ - p_i‖ - R_i|
                let residuals = sensors.map((s, i) => {
                    let dx = pass1.p.x - s.x;
                    let dy = pass1.p.y - s.y;
                    let dist = Math.sqrt(dx*dx + dy*dy);
                    return { index: i, resid: Math.abs(dist - distances[i]) };
                });

                // Ordenamiento O(N log N)
                let sortedResid = [...residuals].sort((a, b) => a.resid - b.resid);
                
                // median_r = sorted(resid)[N//2]
                let median_r = sortedResid[Math.floor(sortedResid.length / 2)].resid;

                // threshold = max(3·median_r, 20m)
                // (Escalado para nuestra malla local de +/- 8: 20m equivale aprox a 2.0 unidades)
                let threshold = Math.max(3 * median_r, 2.0);

                // Keep AP_i if resid_i <= threshold
                let keepIndices = residuals.filter(r => r.resid <= threshold).map(r => r.index);

                // Fallback de geometría: >=3 APs always kept
                if (keepIndices.length < 3) {
                    keepIndices = sortedResid.slice(0, 3).map(r => r.index);
                }

                let culledCount = sensors.length - keepIndices.length;

                // Si no hay outliers destructivos, retornar Pass 1 con GDOP y Topología O(1)
                if (culledCount === 0) {
                    let gdop = this.calculateGDOP(sensors, pass1.p);
                    let topology = this.classifyGeometry(sensors);
                    let coupling = this.applyBilateralCoupling(sensors, distances, pass1.p);
                    return { ...pass1, passes: 1, culled: 0, median_r, keptSensors: sensors.length, gdop, topology, coupledPairs: coupling.coupledPairs };
                }

                // PASS 2: Repetir NLS con el set podado (Pruned Set)
                let prunedSensors = keepIndices.map(i => sensors[i]);
                let prunedDistances = keepIndices.map(i => distances[i]);
                
                // 🚀 Aplicar el Acoplamiento Bilateral en los nodos limpios para estabilizar convergencia
                let coupling = this.applyBilateralCoupling(prunedSensors, prunedDistances, pass1.p);
                
                // Usamos las distancias relajadas (coupledDistances) para la pasada definitiva
                let pass2 = this.solvePass(prunedSensors, coupling.coupledDistances, pass1.p);
                
                let gdop = this.calculateGDOP(prunedSensors, pass2.p);
                let topology = this.classifyGeometry(prunedSensors);

                return { 
                    p: pass2.p, 
                    iter: pass1.iter + pass2.iter, 
                    error: pass2.error, 
                    passes: 2, 
                    culled: culledCount, 
                    median_r, 
                    keptSensors: keepIndices.length,
                    gdop: gdop,
                    topology: topology,
                    coupledPairs: coupling.coupledPairs
                };
            }
        }

        // ═══════════════════════════════════════════════════════════
