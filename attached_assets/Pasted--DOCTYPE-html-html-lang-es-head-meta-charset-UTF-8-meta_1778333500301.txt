        // ═══════════════════════════════════════════════════════════
        // 🚀 KERNEL: LEVENBERG-MARQUARDT TRILATERATION ENGINE
        // ═══════════════════════════════════════════════════════════
        class LMEngine {
            static calculateGDOP(sensors, p) {
                let a = 0, b = 0, d = 0;
                for (let i = 0; i < sensors.length; i++) {
                    let dx = p.x - sensors[i].x; let dy = p.y - sensors[i].y;
                    let dist = Math.sqrt(dx*dx + dy*dy) || 0.0001;
                    let u = dx / dist; let v = dy / dist; 
                    a += u * u; b += u * v; d += v * v;
                }
                let det = (a * d) - (b * b);
                if (Math.abs(det) < 1e-10) return 99.0; 
                let traceInv = (a + d) / det;
                return traceInv > 0 ? Math.sqrt(traceInv) : 99.0;
            }

            static classifyGeometry(sensors) {
                if (sensors.length < 3) return { shape: 'Unknown', quality: 0.0 };
                let s0 = sensors[0], s1 = sensors[1], s2 = sensors[2];
                let d01 = Math.sqrt(Math.pow(s0.x - s1.x, 2) + Math.pow(s0.y - s1.y, 2));
                let d02 = Math.sqrt(Math.pow(s0.x - s2.x, 2) + Math.pow(s0.y - s2.y, 2));
                let d12 = Math.sqrt(Math.pow(s1.x - s2.x, 2) + Math.pow(s1.y - s2.y, 2));
                let sides = [d01, d02, d12].sort((a, b) => a - b);
                let ratio = sides[2] / (sides[0] || 0.0001); 
                if (ratio < 2.0) return { shape: 'Isosceles', quality: 1.0 };
                if (ratio <= 4.0) return { shape: 'Scalene', quality: 0.6 };
                return { shape: 'Cross', quality: 0.3 };
            }

            static applyBilateralCoupling(sensors, distances, p_est) {
                let n = sensors.length; let adj = new Array(n).fill(0);
                let err = new Array(n).fill(0); let theta = new Array(n).fill(0);
                const kappa = 0.3; let coupledPairs = 0;
                for (let i = 0; i < n; i++) {
                    let dx = p_est.x - sensors[i].x; let dy = p_est.y - sensors[i].y;
                    err[i] = Math.sqrt(dx*dx + dy*dy) - distances[i];
                    theta[i] = Math.atan2(dy, dx);
                }
                for (let i = 0; i < n; i++) {
                    for (let j = i + 1; j < n; j++) {
                        let dTheta = Math.abs(theta[i] - theta[j]);
                        if (dTheta > Math.PI) dTheta = 2 * Math.PI - dTheta;
                        if (dTheta > 2.5) { 
                            let avg_err = (err[i] + err[j]) / 2.0;
                            adj[i] += kappa * (avg_err - err[i]); adj[j] += kappa * (avg_err - err[j]);
                            coupledPairs++;
                        }
                    }
                }
                let coupledDistances = distances.map((d, i) => d + adj[i]);
                return { coupledDistances, coupledPairs };
            }

            static quantizeBase20(val, min = -10, max = 10) {
                let norm = (val - min) / (max - min);
                let level = Math.max(0, Math.min(19, Math.floor(norm * 20))); 
                let dequant = min + ((level + 0.5) / 20.0) * (max - min);
                return { level, dequant, error: Math.abs(val - dequant) };
            }

            static solvePass(sensors, distances, initialGuess = {x: 0, y: 0}) {
                let p = { x: initialGuess.x, y: initialGuess.y };
                let lambda = 0.1; const maxIter = 50; const tol = 0.01; 
                let normDp = 999; let iter = 0;
                for (iter = 0; iter < maxIter; iter++) {
                    let r = []; let J = []; 
                    for (let i = 0; i < sensors.length; i++) {
                        let dx = p.x - sensors[i].x; let dy = p.y - sensors[i].y;
                        let dist = Math.sqrt(dx*dx + dy*dy) || 0.0001;
                        r.push(dist - distances[i]); J.push([dx/dist, dy/dist]);
                    }
                    let JTJ = [[0, 0], [0, 0]]; let JTr = [0, 0];
                    for (let i = 0; i < sensors.length; i++) {
                        JTJ[0][0] += J[i][0] * J[i][0]; JTJ[0][1] += J[i][0] * J[i][1];
                        JTJ[1][0] += J[i][1] * J[i][0]; JTJ[1][1] += J[i][1] * J[i][1];
                        JTr[0] += J[i][0] * r[i]; JTr[1] += J[i][1] * r[i];
                    }
                    JTJ[0][0] += lambda; JTJ[1][1] += lambda;
                    let det = JTJ[0][0]*JTJ[1][1] - JTJ[0][1]*JTJ[1][0];
                    if (Math.abs(det) < 1e-10) break; 
                    let invJTJ = [[JTJ[1][1]/det, -JTJ[0][1]/det], [-JTJ[1][0]/det, JTJ[0][0]/det]];
                    let dpX = invJTJ[0][0]*JTr[0] + invJTJ[0][1]*JTr[1];
                    let dpY = invJTJ[1][0]*JTr[0] + invJTJ[1][1]*JTr[1];
                    p.x -= dpX; p.y -= dpY;
                    normDp = Math.sqrt(dpX*dpX + dpY*dpY);
                    if (normDp < tol) break;
                    lambda = Math.max(lambda / 2, 1e-6);
                }
                return { p, iter: iter+1, error: normDp };
            }

            static solveRobust(sensors, distances) {
                let pass1 = this.solvePass(sensors, distances);
                let residuals = sensors.map((s, i) => {
                    let dx = pass1.p.x - s.x; let dy = pass1.p.y - s.y;
                    return { index: i, resid: Math.abs(Math.sqrt(dx*dx + dy*dy) - distances[i]) };
                });
                let sortedResid = [...residuals].sort((a, b) => a.resid - b.resid);
                let median_r = sortedResid[Math.floor(sortedResid.length / 2)].resid;
                let threshold = Math.max(3 * median_r, 2.0); // 2.0m base for small grid
                
                let keepIndices = residuals.filter(r => r.resid <= threshold).map(r => r.index);
                if (keepIndices.length < 3) keepIndices = sortedResid.slice(0, 3).map(r => r.index);
                
                let culledCount = sensors.length - keepIndices.length;
                if (culledCount === 0) {
                    let coupling = this.applyBilateralCoupling(sensors, distances, pass1.p);
                    return { ...pass1, passes: 1, culled: 0, median_r, keptSensors: sensors.length, gdop: this.calculateGDOP(sensors, pass1.p), topology: this.classifyGeometry(sensors), coupledPairs: coupling.coupledPairs };
                }

                let prunedSensors = keepIndices.map(i => sensors[i]);
                let prunedDistances = keepIndices.map(i => distances[i]);
                let coupling = this.applyBilateralCoupling(prunedSensors, prunedDistances, pass1.p);
                let pass2 = this.solvePass(prunedSensors, coupling.coupledDistances, pass1.p);
                
                return { p: pass2.p, iter: pass1.iter + pass2.iter, error: pass2.error, passes: 2, culled: culledCount, median_r, keptSensors: keepIndices.length, gdop: this.calculateGDOP(prunedSensors, pass2.p), topology: this.classifyGeometry(prunedSensors), coupledPairs: coupling.coupledPairs };
            }
        }
